// Logs every submission to a "Submissions" sheet // Columns: Timestamp | Name | Roll No | Email | Role | Task | Answer | Grok Score | Feedback | Files function doPost(e) { const d = JSON.parse(e.postData.contents); const ss = SpreadsheetApp.openById('YOUR_SHEET_ID'); let sh = ss.getSheetByName('Submissions') || ss.insertSheet('Submissions'); if (sh.getLastRow() === 0) sh.appendRow(['Timestamp','Name','Roll No','Email','Role', 'Task','Answer','Grok Score','Feedback','Files']); sh.appendRow([new Date(), d.name, d.roll, d.email||'', d.role, d.task, d.answer, d.score, d.feedback, d.files||'']); return ContentService .createTextOutput(JSON.stringify({status:'ok'})) .setMimeType(ContentService.MimeType.JSON); }

// Expert evaluation model — extend your existing evaluation Sheet // Receives: {name, roll, email, role, task, taskKey, answer, modelAnswer} // Returns: {feedback, scores:{legal_reasoning, evidence_citation, structure, overall}} function doPost(e) { const d = JSON.parse(e.postData.contents); const answer = (d.answer || '').toLowerCase(); const model = (d.modelAnswer || '').toLowerCase(); // ── YOUR EXISTING RUBRIC LOGIC ──────────────────────────── // Replace / extend these checks with your actual rubric keywords const taskRubrics = { 'Show Cause Notice': { legal_reasoning: ['pfutp','§3','§4','fraudulent','manipulat'], evidence_citation: ['exhibit','exhibit b','exhibit c','exhibit e','demat'], structure: ['show cause','opportunity','reply deadline','contemplated'] }, 'Reply to SCN': { legal_reasoning: ['textile','sector rally','causation','disclaimer','good faith'], evidence_citation: ['exhibit e','@stockalertindia','28 march','textile index'], structure: ['preliminary','substantive','relief','personal hearing'] }, 'Investigation Report': { legal_reasoning: ['prima facie','pfutp','ia regulations','recommend'], evidence_citation: ['exhibit b','exhibit c','exhibit d','exhibit e','ip address'], structure: ['background','evidence analysis','findings','recommendation'] }, 'Enforcement Presentation': { legal_reasoning: ['kanaiyalal','shailesh soni','balance of probabilities','disgorgement'], evidence_citation: ['exhibit b','exhibit c','4x','sector index'], structure: ['opening','rebuttal','penalty prayer','debarment'] }, 'Defence Arguments': { legal_reasoning: ['causation','admissibility','manipulative intent','mitigation'], evidence_citation: ['exhibit e','@stockalertindia','bse textile','capital gains'], structure: ['opening','challenge','causation','penalty'] }, 'Adjudication Order': { legal_reasoning: ['§15j','disproportionate gain','investor loss','preponderance'], evidence_citation: ['exhibit b','exhibit c','exhibit e','kanaiyalal'], structure: ['finding','reasoning','penalty','disgorgement','debarment'] } }; const rubric = taskRubrics[d.task] || taskRubrics['Show Cause Notice']; function score(keywords) { const hits = keywords.filter(k => answer.includes(k)).length; const base = Math.round((hits / keywords.length) * 7) + 2; // 2–9 range return Math.min(10, Math.max(1, base)); } const lr = score(rubric.legal_reasoning); const ec = score(rubric.evidence_citation); const st = score(rubric.structure); const ov = Math.round((lr + ec + st) / 3 * 10) / 10; // Build feedback string const missing = []; rubric.legal_reasoning.filter(k=>!answer.includes(k)).slice(0,2).forEach(k=>missing.push(k)); rubric.evidence_citation.filter(k=>!answer.includes(k)).slice(0,1).forEach(k=>missing.push(k)); const feedbackParts = []; if (lr >= 7) feedbackParts.push('Legal reasoning is well-grounded.'); else feedbackParts.push('Legal reasoning needs stronger provision citation.'); if (ec >= 7) feedbackParts.push('Evidence is cited effectively.'); else feedbackParts.push('More specific exhibit references would strengthen this submission.'); if (missing.length) feedbackParts.push(`Key elements to add: ${missing.join(', ')}.`); const result = { feedback: feedbackParts.join(' '), scores: { legal_reasoning: lr, evidence_citation: ec, structure: st, overall: ov } }; // ── LOG TO SHEET ────────────────────────────────────────── const ss = SpreadsheetApp.openById('YOUR_SHEET_ID'); let sh = ss.getSheetByName('Expert Evaluations') || ss.insertSheet('Expert Evaluations'); if (sh.getLastRow() === 0) sh.appendRow(['Timestamp','Name','Roll No','Email','Role','Task', 'LR Score','EC Score','ST Score','Overall','Feedback']); sh.appendRow([new Date(), d.name, d.roll, d.email||'', d.role, d.task, lr, ec, st, ov, result.feedback]); return ContentService .createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); }
